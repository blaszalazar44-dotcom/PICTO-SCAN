// Procesador de texto mejorado: Limpia acentos, pasa a mayúsculas y busca en ARASAAC async function processText() { const rawText = document.getElementById('manualText').value; const upperText = rawText.toUpperCase(); document.getElementById('uppercaseOutput').innerText = upperText || "-"; const container = document.getElementById('pictogramsContainer'); container.innerHTML = ""; if (!rawText.trim()) { container.innerHTML = '

Los pictogramas aparecerán aquí al ingresar o escanear texto...

'; return; } // Separar el texto en palabras const words = rawText.trim().split(/\s+/); for (let word of words) { // Limpiamos signos de puntuación let cleanWord = word.replace(/[.,\/#$%\^&\*;:{}=\-_`~()?¿¡]/g, "").toLowerCase(); if (!cleanWord) continue; // Función para quitar acentos (para que la API no falle con á, é, í, ó, ú, ñ) const normalizedWord = cleanWord.normalize("NFD").replace(/[\u0300-\u036f]/g, ""); // Creamos la tarjeta visual para cada pictograma const card = document.createElement('div'); card.className = "flex flex-col items-center bg-white border border-slate-200 rounded-xl p-2 shadow-xs w-24"; const img = document.createElement('img'); img.className = "w-16 h-16 object-contain mb-1"; img.alt = cleanWord; // Usamos la palabra normalizada (sin tildes) para asegurar que la API la encuentre img.src = `https://api.arasaac.org/api/pictograms/${encodeURIComponent(normalizedWord)}?download=false`; // Si la API no encuentra el pictograma, muestra un ícono de alerta o texto alternativo img.onerror = function() { this.src = "https://api.arasaac.org/api/pictograms/interrogation?download=false"; }; const label = document.createElement('span'); label.className = "text-xs font-bold text-slate-700 text-center truncate w-full uppercase"; label.innerText = cleanWord; card.appendChild(img); card.appendChild(label); container.appendChild(card); } }
